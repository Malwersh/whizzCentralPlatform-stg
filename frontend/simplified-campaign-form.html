Advanced Targeting Rules Examples

New Customers Only:

{
  "customerAge": {"maxDays": 7},
  "orderHistory": {"maxOrders": 0}
}

High-Value Returning Customers:

{
  "orderHistory": {"minOrders": 5},
  "totalSpent": {"minAmount": 500},
  "lastOrder": {"maxDaysAgo": 30}
}

Location-Based Targeting:

{
  "location": {
    "cities": ["Dubai", "Abu Dhabi"],
    "radius": {"center": "25.2048,55.2708", "km": 10}
  }
}

{
  "timeWindow": {
    "hours": [18, 19, 20, 21],
    "weekdays": ["friday", "saturday"]
  }
}
